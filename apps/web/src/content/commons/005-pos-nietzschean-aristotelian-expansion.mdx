## Volitional Fusion Schema: RRF Implementation

Hybrid Aristotelian-Nietzschean fusion using Reciprocal Rank Fusion for optimal virtue-will matching.

```typescript
// Volitional Fusion Schema
interface VolitionalFusionResult {
  virtue: ExpandedVirtue;
  will: VolitionalVirtue;
  rrfScore: number;
  fusedAction: string;
  eudaimonicGain: number;
  übermenschMetric: number;
}

function fuseVirtueWill(
  virtueResults: Array<{ virtue: ExpandedVirtue; score: number }>,
  willResults: Array<{ will: VolitionalVirtue; score: number }>,
  k: number = 60
): VolitionalFusionResult[] {
  // RRF: score = Σ 1/(k + rank)
  const rrfScores = new Map<string, number>();

  // Rank virtues by Aristotelian mean
  virtueResults.forEach((v, index) => {
    const key = `${v.virtue}`;
    const currentScore = rrfScores.get(key) || 0;
    rrfScores.set(key, currentScore + 1 / (k + index + 1));
  });

  // Rank wills by Nietzschean affirmation
  willResults.forEach((w, index) => {
    const key = `${w.will}`;
    const currentScore = rrfScores.get(key) || 0;
    rrfScores.set(key, currentScore + 1 / (k + index + 1));
  });

  // Return fused results sorted by RRF score
  return Array.from(rrfScores.entries())
    .sort((a, b) => b[1] - a[1])
    .map(([key, score]) => ({
      virtue: key as ExpandedVirtue,
      will: key as VolitionalVirtue,
      rrfScore: score,
      fusedAction: `Execute with mean-will balance`,
      eudaimonicGain: score * 100,
      übermenschMetric: score * 0.8
    }));
}
```

## Rate Limiting: Token Bucket for Volitional Calls

```typescript
interface VolitionalBucket {
  tokens: number;
  maxTokens: number;
  refillRate: number; // tokens per hour
  lastRefill: Date;
}

class VolitionalRateLimiter {
  private bucket: VolitionalBucket;

  constructor(maxDaily: number = 4) {
    this.bucket = {
      tokens: maxDaily,
      maxTokens: maxDaily,
      refillRate: maxDaily / 8, // Refill over 8 waking hours
      lastRefill: new Date()
    };
  }

  canFuse(): boolean {
    this.refill();
    return this.bucket.tokens >= 1;
  }

  consumeFusion(): boolean {
    if (!this.canFuse()) {
      console.warn('Volitional rate limit exceeded. Wait for refill.');
      return false;
    }
    this.bucket.tokens -= 1;
    return true;
  }

  private refill(): void {
    const now = new Date();
    const hoursSinceRefill =
      (now.getTime() - this.bucket.lastRefill.getTime()) / (1000 * 60 * 60);
    const tokensToAdd = Math.floor(hoursSinceRefill * this.bucket.refillRate);

    if (tokensToAdd > 0) {
      this.bucket.tokens = Math.min(
        this.bucket.maxTokens,
        this.bucket.tokens + tokensToAdd
      );
      this.bucket.lastRefill = now;
    }
  }
}
```

## Error Handling: Volitional Rollback Protocol

```typescript
interface VolitionalState {
  currentVirtue: ExpandedVirtue;
  currentWill: VolitionalVirtue;
  meanBalance: number;
  willPower: number;
  timestamp: Date;
}

class VolitionalRollback {
  private stateHistory: VolitionalState[] = [];
  private maxHistorySize: number = 10;

  saveState(state: VolitionalState): void {
    this.stateHistory.push({ ...state, timestamp: new Date() });
    if (this.stateHistory.length > this.maxHistorySize) {
      this.stateHistory.shift();
    }
  }

  rollback(): VolitionalState | null {
    if (this.stateHistory.length < 2) {
      console.warn('No previous state to rollback to.');
      return null;
    }

    // Remove current state
    this.stateHistory.pop();

    // Return to previous stable state
    const previousState = this.stateHistory[this.stateHistory.length - 1];
    console.log(`Rolling back to Aristotelian mean state: ${previousState.currentVirtue}`);

    return previousState;
  }

  handleImbalance(error: Error, currentState: VolitionalState): VolitionalState {
    console.error(`Volitional imbalance detected: ${error.message}`);

    // Log for phronesis audit
    console.log(`Audit log: ${JSON.stringify({
      error: error.message,
      state: currentState,
      timestamp: new Date().toISOString()
    })}`);

    // Attempt rollback
    const rolledBackState = this.rollback();

    if (rolledBackState) {
      return rolledBackState;
    }

    // Default to balanced Aristotelian mean
    return {
      currentVirtue: 'temperance',
      currentWill: 'self_overcoming',
      meanBalance: 0.5,
      willPower: 0.5,
      timestamp: new Date()
    };
  }
}
```
